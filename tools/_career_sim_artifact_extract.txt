<script>

const API_URL = "https://api.anthropic.com/v1/messages";
const MODEL = "claude-sonnet-4-6";

const STAGES = ["intake","confirm","menu","simulation","selfassess","debrief"];
let stage = "intake";
let jd = { raw:"", title:"", org:"", responsibilities:[], skills:[] };
let scenarios = [];
let chosenScenario = null;
let history = []; // {role, content}
let exchangeCount = 0;
let selfRatings = {}; // skill -> 1-5
let reflectionText = "";
let attempted = [];

const app = document.getElementById('app');
const sceneNum = document.getElementById('sceneNum');
const progressEl = document.getElementById('progress');

function renderProgress(){
  progressEl.innerHTML = STAGES.map(s=>{
    const idx = STAGES.indexOf(stage);
    const i = STAGES.indexOf(s);
    return `<div class="${i<=idx?'done':''}"></div>`;
  }).join('');
}

function setStage(s, label){
  stage = s;
  sceneNum.textContent = "SCENE — " + (label || s).toUpperCase();
  renderProgress();
}


async function callClaude(systemPrompt, messages, opts={}){
  const body = {
    model: MODEL,
    max_tokens: 1000,
    system: systemPrompt,
    messages: messages
  };
  const res = await fetch(API_URL, {
    method:"POST",
    headers:{"Content-Type":"application/json"},
    body: JSON.stringify(body)
  });
  const data = await res.json();
  const text = (data.content||[]).filter(b=>b.type==="text").map(b=>b.text).join("\n");
  return text;
}

function extractJSON(text){
  const cleaned = text.replace(/```json|```/g,"").trim();
  const start = cleaned.indexOf('{') === -1 ? cleaned.indexOf('[') : cleaned.indexOf('{');
  const tryParse = (s)=>{ try{ return JSON.parse(s); }catch(e){ return null; } };
  let parsed = tryParse(cleaned);
  if(parsed) return parsed;
  // fallback: slice from first brace/bracket to last matching one
  const lastObj = cleaned.lastIndexOf('}');
  const lastArr = cleaned.lastIndexOf(']');
  const end = Math.max(lastObj, lastArr);
  if(start>=0 && end>start) parsed = tryParse(cleaned.slice(start, end+1));
  return parsed;
}

function loadingHTML(messages){

  return `<div class="card"><div class="loading"><div class="dot"></div><div class="dot"></div><div class="dot"></div>&nbsp; ${messages}</div></div>`;
}

/* ---------------- STAGE 1: INTAKE ---------------- */
function renderIntake(){
  setStage("intake","Intake");
  app.innerHTML = `
    <div class="card">
      <h2>Paste the job description</h2>
      <p style="color:var(--ink-dim); font-size:14px; margin-top:-6px;">Any education-related role — teacher, instructional designer, curriculum developer, program coordinator, school counselor, EdTech specialist. We'll build the rehearsal around it.</p>
      <span class="label">Job Description</span>
      <textarea id="jdInput" placeholder="Paste the full job posting here..."></textarea>
      <div class="row">
        <button class="btn-primary" id="parseBtn">Read the script</button>
      </div>
    </div>
  `;
  document.getElementById('parseBtn').onclick = async ()=>{
    const val = document.getElementById('jdInput').value.trim();
    if(!val) return;
    jd.raw = val;
    app.innerHTML = loadingHTML("Reading the job description");
    const sys = `You extract structured information from job descriptions for a career-simulation tool. Respond ONLY with valid JSON, no preamble, no markdown fences. Schema:
{"title": string, "orgType": string, "responsibilities": [4 strings], "skills": [5 strings, pulled close to verbatim from the JD]}`;
    const text = await callClaude(sys, [{role:"user", content: val}]);
    const parsed = extractJSON(text);
    if(parsed){
      jd.title = parsed.title; jd.org = parsed.orgType;
      jd.responsibilities = parsed.responsibilities||[];
      jd.skills = parsed.skills||[];
    }
    renderConfirm();

  };
}

/* ---------------- STAGE 2: CONFIRM ---------------- */
function renderConfirm(){
  setStage("confirm","Confirm Casting");
  document.getElementById('headline').textContent = jd.title || "Career Simulation";
  app.innerHTML = `
    <div class="card">
      <h2>Role: ${jd.title || '—'} <span style="color:var(--ink-faint); font-weight:400;">at ${jd.org || '—'}</span></h2>
      <span class="label">Core responsibilities</span>
      <ul class="bullets">${jd.responsibilities.map(r=>`<li>${r}</li>`).join('')}</ul>
      <span class="label" style="margin-top:16px;">Skills this rehearsal will assess</span>
      <div>${jd.skills.map(s=>`<span class="skill-chip">${s}</span>`).join('')}</div>
      <div class="row">
        <button class="btn-primary" id="confirmBtn">This looks right — build the scenarios</button>
        <button class="btn-ghost" id="backBtn">Start over</button>
      </div>
    </div>
  `;
  document.getElementById('backBtn').onclick = renderIntake;
  document.getElementById('confirmBtn').onclick = async ()=>{
    app.innerHTML = loadingHTML("Drafting four scenes for this role");
    const sys = `You design realistic workplace roleplay scenarios for soon-to-graduate students based on a specific job description. Generate exactly 4 scenarios tailored to this JD — never generic, every scenario must reference something concrete from the responsibilities. Respond ONLY with valid JSON, no preamble:
[{"title": string, "description": string (2 sentences)}, ... x4]`;
    const userMsg = `Job title: ${jd.title}\nOrg type: ${jd.org}\nResponsibilities: ${jd.responsibilities.join('; ')}\nSkills: ${jd.skills.join('; ')}`;
    const text = await callClaude(sys, [{role:"user", content:userMsg}]);
    const parsed = extractJSON(text);
    scenarios = parsed || [];
    renderMenu();
  };
}


/* ---------------- STAGE 3: MENU ---------------- */
function renderMenu(){
  setStage("menu","Choose A Scene");
  app.innerHTML = `
    <div class="card">
      <h2>Pick a scenario to step into</h2>
      ${scenarios.map((s,i)=>`
        <div class="scenario" data-i="${i}">
          <div class="num">SCENE ${i+1}${attempted.includes(i)?' · revisited':''}</div>
          <h3>${s.title}</h3>
          <p>${s.description}</p>
        </div>
      `).join('')}
    </div>
  `;
  [...document.querySelectorAll('.scenario')].forEach(el=>{
    el.onclick = ()=> startSimulation(parseInt(el.dataset.i), false);
  });
}

/* ---------------- STAGE 4: SIMULATION ---------------- */
async function startSimulation(idx, hardMode){
  chosenScenario = scenarios[idx];
  if(!attempted.includes(idx)) attempted.push(idx);
  history = [];
  exchangeCount = 0;
  setStage("simulation","Scene " + (idx+1));
  app.innerHTML = loadingHTML("Setting the scene");

  const sys = `You are running a career-simulation roleplay for someone training for: ${jd.title} at ${jd.org}.
Scenario: "${chosenScenario.title}" — ${chosenScenario.description}

Skills being assessed: ${jd.skills.join(', ')}.
${hardMode ? 'HARD MODE: play the character as notably more resistant, skeptical, or difficult than a baseline version.' : ''}

First message only: respond with JSON: {"setScene": "3-4 vivid sentences describing situation, setting, moment before dialogue begins", "characterIntro": "1-2 sentences, first person from you, e.g. I'm playing [Name], a [role/description], explaining their perspective and what they need to be convinced of", "openingLine": "the character's first line of dialogue, natural and realistic"}
Respond ONLY with that JSON, no markdown fences.`;

  const text = await callClaude(sys, [{role:"user", content:"Begin the simulation."}]);
  const parsed = extractJSON(text);
  history.push({role:"assistant", content: JSON.stringify(parsed)});

  renderSimUI(parsed.setScene, parsed.characterIntro, parsed.openingLine, hardMode);
}

function renderSimUI(setScene, characterIntro, openingLine, hardMode){
  app.innerHTML = `
    <div class="card">
      <div class="stage-note">${setScene}</div>
      <div class="bubble char"><span class="who">Your role for this scene</span>${characterIntro}</div>
      <div id="thread"></div>
      <div class="input-row">
        <input type="text" id="userMsg" placeholder="Type your response...">
        <button class="btn-primary" id="sendBtn">Respond</button>
      </div>
      <div class="row">
        <button class="btn-ghost" id="endBtn">End simulation</button>
      </div>
    </div>
  `;
  appendBubble('char', openingLine);
  document.getElementById('sendBtn').onclick = ()=>sendUserLine(hardMode);
  document.getElementById('userMsg').addEventListener('keydown', e=>{ if(e.key==='Enter') sendUserLine(hardMode); });
  document.getElementById('endBtn').onclick = ()=>endSimulation(hardMode);

}

function appendBubble(who, text){
  const thread = document.getElementById('thread');
  const div = document.createElement('div');
  div.className = 'bubble ' + (who==='char' ? 'char' : 'user');
  div.innerHTML = `<span class="who">${who==='char' ? 'Character' : 'You'}</span>${text}`;
  thread.appendChild(div);
  window.scrollTo({top: document.body.scrollHeight, behavior:'smooth'});
}

async function sendUserLine(hardMode){
  const input = document.getElementById('userMsg');
  const val = input.value.trim();
  if(!val) return;
  input.value = '';
  appendBubble('user', val);
  if(val.toLowerCase() === 'end simulation'){ return endSimulation(hardMode); }

  exchangeCount++;
  document.getElementById('sendBtn').disabled = true;

  const sys = `Continue playing the character from this career-simulation scenario in-role. Scenario: "${chosenScenario.title}" — ${chosenScenario.description}. Skills under assessment: ${jd.skills.join(', ')}.
${hardMode ? 'HARD MODE: character remains notably resistant/skeptical.' : ''}
Stay fully in character — never break character or coach, even if asked "how am I doing". Keep your reply to 2-4 natural sentences. If the user's response is short or vague, press for specifics. If they seem stuck, the character can offer an opening. Reply with plain dialogue text only, no JSON, no stage directions in brackets.`;

  history.push({role:"user", content: val});
  const claudeMessages = history.map(h=>({role:h.role, content:h.content}));
  const reply = await callClaude(sys, claudeMessages);
  history.push({role:"assistant", content: reply});
  appendBubble('char', reply);
  document.getElementById('sendBtn').disabled = false;


  if(exchangeCount >= 10){
    appendBubble('char', "(The scene has run long — let's pause here.)");
    setTimeout(()=>endSimulation(hardMode), 600);
  }
}

function endSimulation(hardMode){
  renderSelfAssess();
}

/* ---------------- STAGE 5: SELF-ASSESSMENT (NEW) ---------------- */
function renderSelfAssess(){
  setStage("selfassess","Self-Assessment");
  selfRatings = {};
  app.innerHTML = `
    <div class="card">
      <h2>Before we debrief — rate yourself</h2>
      <p style="color:var(--ink-dim); font-size:14px; margin-top:-6px;">No one's grading this. It just helps you notice the gap (or overlap) between how it felt in the moment and what actually came across.</p>
      ${jd.skills.map((s,i)=>`
        <div class="skill-rate">
          <div class="sk-name">${s}</div>
          <div class="stars" data-skill="${i}">
            ${[1,2,3,4,5].map(n=>`<div class="star" data-val="${n}">${n}</div>`).join('')}
          </div>
        </div>
      `).join('')}
      <span class="label" style="margin-top:18px;">In your own words — how do you think that went?</span>
      <textarea id="reflection" placeholder="Optional: what felt strong, what felt shaky..." style="min-height:90px;"></textarea>
      <div class="row">
        <button class="btn-primary" id="toDebrief">See the debrief</button>
        <button class="btn-ghost" id="skipAssess">Skip self-assessment</button>

      </div>
    </div>
  `;
  document.querySelectorAll('.stars').forEach(group=>{
    const skillIdx = group.dataset.skill;
    group.querySelectorAll('.star').forEach(star=>{
      star.onclick = ()=>{
        const val = parseInt(star.dataset.val);
        selfRatings[jd.skills[skillIdx]] = val;
        group.querySelectorAll('.star').forEach(s=>{
          s.classList.toggle('on', parseInt(s.dataset.val) <= val);
        });
      };
    });
  });
  document.getElementById('toDebrief').onclick = ()=>{
    reflectionText = document.getElementById('reflection').value.trim();
    runDebrief();
  };
  document.getElementById('skipAssess').onclick = ()=>{
    selfRatings = {}; reflectionText = '';
    runDebrief();
  };
}

/* ---------------- STAGE 6: DEBRIEF ---------------- */
async function runDebrief(){
  setStage("debrief","Notes Session");
  app.innerHTML = loadingHTML("Pulling notes together");

  const transcript = history.map(h=>{
    if(h.role==='user') return "USER: " + h.content;

    try{
      const parsed = JSON.parse(h.content);
      if(parsed.openingLine) return "CHARACTER (opening): " + parsed.openingLine;
    }catch(e){}
    return "CHARACTER: " + h.content;
  }).join('\n');

  const selfAssessSummary = Object.keys(selfRatings).length
    ? jd.skills.map(s=> `${s}: self-rated ${selfRatings[s]||'n/a'}/5`).join('; ') + (reflectionText ? `\nUser reflection: "${reflectionText}"` : '')
    : 'User skipped self-assessment.';

  const sys = `You are debriefing a career-simulation roleplay. Be warm and encouraging in tone, but honest and specific — avoid hollow praise. Reference actual moments from the transcript. Tie feedback directly to the listed JD skills, not generic soft-skill labels. The user has also self-assessed before seeing your notes; where their self-rating and your read diverge, gently name that gap as useful information, not as a gotcha.

Respond ONLY with valid JSON, no markdown fences, schema:
{
  "summary": "3-4 neutral sentences on what happened",
  "skillEvals": [ {"skill": string, "observed": "1 sentence on what was observed", "stronger": "1 sentence on what could be stronger or what they did well", "selfGap": "1 short sentence noting alignment or gap with their self-rating, or empty string if they skipped self-assessment"} ],
  "tryNext": "one concrete, actionable technique to try on a retry, framed as a technique not a critique"
}`;

  const userMsg = `Role: ${jd.title} at ${jd.org}\nScenario: ${chosenScenario.title} — ${chosenScenario.description}\nSkills: ${jd.skills.join('; ')}\n\nTranscript:\n${transcript}\n\nSelf-assessment:\n${selfAssessSummary}`;

  const text = await callClaude(sys, [{role:"user", content:userMsg}]);
  const parsed = extractJSON(text) || {summary:text, skillEvals:[], tryNext:""};
  renderDebrief(parsed);
}

function renderDebrief(d){
  app.innerHTML = `
    <div class="card">
      <h2>Scene. Let's debrief.</h2>
      <div class="debrief-block">

        <h3>What happened</h3>
        <p style="color:var(--ink-dim); line-height:1.6;">${d.summary}</p>
      </div>
      <div class="debrief-block">
        <h3>Skills from the JD — how it went</h3>
        ${(d.skillEvals||[]).map(s=>`
          <div class="skill-eval">
            <div class="sk-head">
              <b>${s.skill}</b>
              ${s.selfGap ? `<span class="self-vs">${s.selfGap}</span>` : ''}
            </div>
            <p>${s.observed} ${s.stronger}</p>
          </div>
        `).join('')}
      </div>
      <div class="debrief-block">
        <h3>One thing to try on the retry</h3>
        <div class="try-box">${d.tryNext}</div>
      </div>
      <div class="row" style="margin-top:8px;">
        <button class="btn-primary" id="retrySame">Retry — same scene</button>
        <button class="btn-ghost" id="retryHard">Retry — hard mode</button>
        <button class="btn-ghost" id="newScene">Different scenario</button>
      </div>
    </div>
  `;
  const idx = scenarios.indexOf(chosenScenario);
  document.getElementById('retrySame').onclick = ()=> startSimulation(idx, false);
  document.getElementById('retryHard').onclick = ()=> startSimulation(idx, true);
  document.getElementById('newScene').onclick = renderMenu;
}


renderIntake();
